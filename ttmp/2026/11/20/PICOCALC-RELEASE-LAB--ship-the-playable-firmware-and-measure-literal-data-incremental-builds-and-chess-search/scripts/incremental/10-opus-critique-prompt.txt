// Private prototype. Every committed backend-version/ABI identity is supplied by the harness.
globalThis.fragmentSession = function(prior) {
    let descriptor, reusable, priorRoot, reused = 0, rebuilt = 0;
    const next = Object.create(null);
    function init(imports, functions, records, enums, distinct, lowered) {
        descriptor = JSON.stringify([
            imports.map(f => [f.name,f.params,f.result,Boolean(f.awaitable)]),
            functions.map(f => [f.name,f.params,f.paramNames,f.result]),
            [...records].map(([name,fields]) => [name,[...fields]]),
            [...enums].map(([name,fields]) => [name,[...fields]]), [...distinct],
            lowered ? [lowered.prefix,[...lowered.startFields]] : null
        ]);
        priorRoot = prior && prior.descriptor === descriptor ? prior.items : null;
    }
    function lookup(lex, fn, edges) {
        const old = priorRoot && Object.hasOwn(priorRoot,fn.name) && priorRoot[fn.name];
        reusable = null;
        if (old && lex.source.startsWith(old.body,fn.bodyAt)) {
            for (const edge of old.edges) edges.push(edge);
            reusable = next[fn.name] = old;
            reused++;
            return true;
        }
        lex.offset = fn.bodyAt;
        lex.next();
        const end = lex.skipBody();
        if (end !== undefined) next[fn.name] = {body:lex.source.slice(fn.bodyAt,end), edges};
        rebuilt++;
        return false;
    }
    function lower(fn, targets, build) {
        if (reusable) {
            const old = reusable;
            return {words:old.words.slice(), calls:old.calls.map(([from,index,offset,rd]) =>
                ({from, target:index < 0 ? {at:offset} : targets[index], rd}))};
        }
        const built = build();
        const saved = next[fn.name];
        if (saved) {
            saved.words = built.words.slice();
            saved.calls = built.calls.map(({from,target,rd}) => {
                const index = targets.indexOf(target);
                return [from,index,index < 0 ? target.at : 0,rd];
            });
        }
        return built;
    }
    return {init,lookup,lower,commit:() => ({descriptor,items:next}),stats:() => ({reused,rebuilt})};
};

globalThis.incrementalEdits = function(app, source) {
    function change(text, from, to, count=1) {
        if(text.split(from).length!==count+1) throw Error("unexpected source anchor: "+from);
        return text.split(from).join(to);
    }
    if(app==="editor") {
        const body=change(source,"return (row % 36) * 100;","return (row % 36) * 99 + 1;");
        const bad=change(body,"editorPreviewSlot(row: i32)","editorPreviewSlot(row: i32, stride: i32)");
        let signature=change(bad,"* 99 + 1;","* stride + 1;");
        signature=change(signature,"editorPreviewSlot(state.previewRows)","editorPreviewSlot(state.previewRows, 99)",2);
        signature=change(signature,"editorPreviewSlot(row)","editorPreviewSlot(row, 99)");
        const layout=change(signature,"    file: i32;\n    opening: bool;","    opening: bool;\n    file: i32;");
        return {base:source,body,bad,signature,layout};
    }
    const body=change(source,"return chessSquare((move / 64) % 64);","return chessSquare((move / 64) % 64) + 1;");
    const bad=change(body,"function chessFrom(move: i32)","function chessFrom(move: i32, bias: i32)");
    let signature=change(bad,"return chessSquare((move / 64) % 64) + 1;","return chessSquare((move / 64) % 64) + bias;");
    signature=change(signature,"chessFrom(move)","chessFrom(move, 1)",7);
    signature=change(signature,"chessFrom(state.lastMove)","chessFrom(state.lastMove, 1)",3);
    const layout=change(signature,"    cursor: i32;\n    selected: i32;","    selected: i32;\n    cursor: i32;");
    return {base:source,body,bad,signature,layout};
};
